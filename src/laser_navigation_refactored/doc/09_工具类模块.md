# 9. 工具类模块

本章详细介绍项目中的工具类，包括 JSON 解析器、贝塞尔控制点生成器和数学工具。

## 9.1 JSON 解析器（JsonParser）

### 9.1.1 模块概述

`JsonParser` 负责将 JSON 格式的导航任务配置解析为内部数据结构。

### 9.1.2 类定义

```cpp
class JsonParser {
public:
    JsonParser() = default;
    ~JsonParser() = default;

    /// @brief 解析导航配置JSON
    /// @param json_string JSON字符串
    /// @return Optional<NavigationConfig> 解析成功返回配置，失败返回nullopt
    Optional<NavigationConfig> parseNavigationConfig(
        const std::string& json_string);

    /// @brief 解析单个路径点
    /// @param json JSON对象
    /// @return Pose2D 路径点
    Pose2D parseWaypoint(const nlohmann::json& json);

    /// @brief 解析运动约束
    /// @param json JSON对象
    /// @return MotionConstraints 运动约束
    MotionConstraints parseConstraints(const nlohmann::json& json);

    /// @brief 解析底盘类型
    /// @param json JSON对象
    /// @return ChassisType 底盘类型
    ChassisType parseChassisType(const nlohmann::json& json);

    /// @brief 解析路径段类型
    /// @param type_str 类型字符串
    /// @return PathSegmentType 路径段类型
    PathSegmentType parseSegmentType(const std::string& type_str);
    
    /// @brief 获取解析错误信息
    std::string getLastError() const { return last_error_; }

private:
    std::string last_error_;
};
```

### 9.1.3 JSON 格式与解析

#### 底盘类型配置（model.json）

底盘类型在节点启动时从单独的 `model.json` 文件读取：

```json
{
    "agvType": "4WIS4WID"
}
```

- `4WIS4WID`: 四转四驱底盘
- `DiffDrive`: 双轮差速底盘

#### 导航任务JSON格式

```json
{
    "adjust_start_angle": true,
    "adjust_end_angle": true,
    "auto_generate_control_points": false,
    "waypoints": [
        {
            "x": 0.0, "y": 0.0, "yaw": 0.0,
            "type": "Straight",
            "constraints": {
                "max_velocity": 0.5,
                "max_acceleration": 0.3,
                "max_deceleration": 0.3,
                "max_angular_velocity": 0.5,
                "max_angular_acceleration": 0.3,
                "is_forward": true,
                "reach_distance": 0.05,
                "reach_angle": 0.05
            }
        },
        {
            "x": 1.0, "y": 0.0, "yaw": 1.57,
            "type": "CubicBezier",
            "controlPoints": [
                {"x": 0.3, "y": 0.0},
                {"x": 0.7, "y": 0.5}
            ],
            "constraints": { ... }
        },
        {
            "x": 1.5, "y": 0.5, "yaw": -0.5,  // yaw<0: 曲率模式
            "type": "CircularArc"
        },
        {
            "x": 2.0, "y": 1.0, "yaw": 1.57,  // yaw>=0: 三点模式
            "type": "CircularArc",
            "controlPoint": {"x": 1.7, "y": 0.8}  // 弧中间点
        }
    ]
}
```

#### 解析主函数

```cpp
Optional<NavigationConfig> JsonParser::parseNavigationConfig(
    const std::string& json_string) {
    
    try {
        auto json = nlohmann::json::parse(json_string);
        NavigationConfig config;

        // 注意：chassis_type 不在此JSON中，由上层从model.json读取后设置

        // 1. 解析角度调整标志
        if (json.contains("adjust_start_angle")) {
            config.adjust_start_angle = json["adjust_start_angle"].get<bool>();
        }
        if (json.contains("adjust_end_angle")) {
            config.adjust_end_angle = json["adjust_end_angle"].get<bool>();
        }

        // 3. 解析路径点数组
        if (!json.contains("waypoints") || !json["waypoints"].is_array()) {
            last_error_ = "Missing or invalid 'waypoints' array";
            return nullopt;
        }

        for (const auto& wp_json : json["waypoints"]) {
            // 解析位姿
            Pose2D pose = parseWaypoint(wp_json);
            config.waypoints.push_back(pose);

            // 解析段类型
            PathSegmentType seg_type = PathSegmentType::kStraight;
            if (wp_json.contains("type")) {
                seg_type = parseSegmentType(wp_json["type"].get<std::string>());
            }
            config.segment_types.push_back(seg_type);

            // 解析约束
            MotionConstraints constraints;
            if (wp_json.contains("constraints")) {
                constraints = parseConstraints(wp_json["constraints"]);
            }
            config.constraints.push_back(constraints);

            // 解析控制点
            std::vector<Pose2D> cp = parseControlPoints(wp_json, seg_type);
            config.control_points.push_back(cp);
        }

        return Optional<NavigationConfig>(config);
        
    } catch (const nlohmann::json::exception& e) {
        last_error_ = std::string("JSON parse error: ") + e.what();
        return nullopt;
    }
}
```

#### 控制点解析

```cpp
std::vector<Pose2D> JsonParser::parseControlPoints(
    const nlohmann::json& wp_json, 
    PathSegmentType seg_type) {
    
    std::vector<Pose2D> points;
    
    // CubicBezier: 两个控制点
    if (seg_type == PathSegmentType::kCubicBezier) {
        if (wp_json.contains("controlPoints") && 
            wp_json["controlPoints"].is_array()) {
            for (const auto& cp : wp_json["controlPoints"]) {
                points.push_back(Pose2D{
                    cp["x"].get<double>(),
                    cp["y"].get<double>(),
                    0.0
                });
            }
        }
    }
    // CircularArc/Dubins/ReedsShepp: 单个控制点（三点模式）
    else if (seg_type == PathSegmentType::kCircularArc ||
             seg_type == PathSegmentType::kDubins ||
             seg_type == PathSegmentType::kReedsShepp) {
        
        double yaw = wp_json["yaw"].get<double>();
        
        // yaw >= 0: 三点模式，需要controlPoint
        if (yaw >= 0.0 && wp_json.contains("controlPoint")) {
            const auto& cp = wp_json["controlPoint"];
            points.push_back(Pose2D{
                cp["x"].get<double>(),
                cp["y"].get<double>(),
                yaw  // 存储终点航向
            });
        }
        // yaw < 0: 曲率模式，yaw存储曲率值
    }
    
    return points;
}
```

### 9.1.4 段类型解析

```cpp
PathSegmentType JsonParser::parseSegmentType(const std::string& type_str) {
    static const std::unordered_map<std::string, PathSegmentType> type_map = {
        {"Straight",      PathSegmentType::kStraight},
        {"CubicBezier",   PathSegmentType::kCubicBezier},
        {"CircularArc",   PathSegmentType::kCircularArc},
        {"Dubins",        PathSegmentType::kDubins},
        {"ReedsShepp",    PathSegmentType::kReedsShepp}
    };
    
    auto it = type_map.find(type_str);
    if (it != type_map.end()) {
        return it->second;
    }
    
    // 默认返回直线
    return PathSegmentType::kStraight;
}
```

### 9.1.5 底盘类型解析（从model.json）

底盘类型从model.json文件读取，而非导航任务JSON：

```cpp
ChassisType JsonParser::parseModelConfig(const std::string& model_json_path) {
    std::ifstream file(model_json_path);
    if (!file.is_open()) {
        return ChassisType::kDifferential;  // 默认差速
    }
    
    auto json = nlohmann::json::parse(file);
    
    if (!json.contains("agvType")) {
        return ChassisType::kDifferential;
    }
    
    std::string type_str = json["agvType"].get<std::string>();
    
    if (type_str == "4WIS4WID") {
        return ChassisType::kSwerve4WIS4WID;
    }
    
    return ChassisType::kDifferential;  // DiffDrive 或其他值
}
```

## 9.2 贝塞尔控制点生成器（BezierControlPointGenerator）

### 9.2.1 模块概述

自动生成三次贝塞尔曲线的控制点，使生成的曲线满足：
- 端点切线方向与路径点航向一致
- 曲线平滑，曲率变化连续
- 可配置曲线"松紧度"

### 9.2.2 类定义

```cpp
class BezierControlPointGenerator {
public:
    BezierControlPointGenerator();
    ~BezierControlPointGenerator() = default;

    /// @brief 设置控制点距离因子
    /// @param factor 因子值(0~1)，越大曲线越"松"
    void setDistanceFactor(double factor) { distance_factor_ = factor; }

    /// @brief 生成单段贝塞尔曲线的控制点
    /// @param start 起点
    /// @param end 终点
    /// @param constraints 运动约束
    /// @return 两个控制点 [P1, P2]
    std::vector<Pose2D> generateControlPoints(
        const Pose2D& start,
        const Pose2D& end,
        const MotionConstraints& constraints);

    /// @brief 批量生成所有段的控制点
    /// @param waypoints 所有路径点
    /// @param segment_types 每段类型
    /// @param constraints 每段约束
    /// @param existing_points 已有的控制点（优先使用）
    /// @return 所有段的控制点
    std::vector<std::vector<Pose2D>> generateAllControlPoints(
        const std::vector<Pose2D>& waypoints,
        const std::vector<PathSegmentType>& segment_types,
        const std::vector<MotionConstraints>& constraints,
        const std::vector<std::vector<Pose2D>>& existing_points);

private:
    double distance_factor_{0.33};  // 默认1/3
};
```

### 9.2.3 控制点生成算法

```
三次贝塞尔曲线: B(t) = (1-t)³P₀ + 3(1-t)²tP₁ + 3(1-t)t²P₂ + t³P₃

控制点生成原则：
- P₁ = P₀ + d * (cos(θ₀), sin(θ₀))
- P₂ = P₃ - d * (cos(θ₃), sin(θ₃))

其中:
- θ₀: 起点航向
- θ₃: 终点航向
- d = factor * ||P₃ - P₀||: 控制点距离
```

```cpp
std::vector<Pose2D> BezierControlPointGenerator::generateControlPoints(
    const Pose2D& start,
    const Pose2D& end,
    const MotionConstraints& constraints) {
    
    std::vector<Pose2D> control_points(2);
    
    // 计算起终点距离
    double dx = end.x - start.x;
    double dy = end.y - start.y;
    double distance = std::sqrt(dx * dx + dy * dy);
    
    // 控制点距离
    double d = distance_factor_ * distance;
    
    // 处理倒车情况
    double start_heading = start.yaw;
    double end_heading = end.yaw;
    if (!constraints.is_forward) {
        start_heading = math::MathUtils::normalizeAngle(start_heading + M_PI);
        end_heading = math::MathUtils::normalizeAngle(end_heading + M_PI);
    }
    
    // 第一控制点 P1: 沿起点航向方向
    control_points[0].x = start.x + d * std::cos(start_heading);
    control_points[0].y = start.y + d * std::sin(start_heading);
    control_points[0].yaw = 0.0;
    
    // 第二控制点 P2: 沿终点航向反方向
    control_points[1].x = end.x - d * std::cos(end_heading);
    control_points[1].y = end.y - d * std::sin(end_heading);
    control_points[1].yaw = 0.0;
    
    return control_points;
}
```

### 9.2.4 批量生成

```cpp
std::vector<std::vector<Pose2D>> BezierControlPointGenerator::generateAllControlPoints(
    const std::vector<Pose2D>& waypoints,
    const std::vector<PathSegmentType>& segment_types,
    const std::vector<MotionConstraints>& constraints,
    const std::vector<std::vector<Pose2D>>& existing_points) {
    
    std::vector<std::vector<Pose2D>> all_points;
    
    for (size_t i = 0; i < waypoints.size() - 1; ++i) {
        // 如果已有控制点，直接使用
        if (i < existing_points.size() && !existing_points[i].empty()) {
            all_points.push_back(existing_points[i]);
            continue;
        }
        
        // 只为贝塞尔段生成控制点
        if (i < segment_types.size() && 
            segment_types[i] == PathSegmentType::kCubicBezier) {
            
            const auto& constraint = (i < constraints.size()) ? 
                constraints[i] : MotionConstraints();
            
            auto points = generateControlPoints(
                waypoints[i], waypoints[i + 1], constraint);
            all_points.push_back(points);
        } else {
            all_points.push_back({});  // 空的控制点
        }
    }
    
    return all_points;
}
```

### 9.3.3 核心函数实现

#### 角度归一化

```cpp
double MathUtils::normalizeAngle(double angle) {
    while (angle > M_PI) angle -= 2.0 * M_PI;
    while (angle < -M_PI) angle += 2.0 * M_PI;
    return angle;
}

double MathUtils::angleDiff(double angle1, double angle2) {
    return normalizeAngle(angle1 - angle2);
}

double MathUtils::absoluteAngleDiff(double angle1, double angle2) {
    return std::abs(angleDiff(angle1, angle2));
}
```

#### 几何计算

```cpp
double MathUtils::distance(const Pose2D& p1, const Pose2D& p2) {
    double dx = p2.x - p1.x;
    double dy = p2.y - p1.y;
    return std::sqrt(dx * dx + dy * dy);
}

double MathUtils::pointToLineDistance(
    double px, double py,
    double lx1, double ly1, double lx2, double ly2) {
    
    double dx = lx2 - lx1;
    double dy = ly2 - ly1;
    double length = std::sqrt(dx * dx + dy * dy);
    
    if (length < 1e-10) {
        return distance(px, py, lx1, ly1);
    }
    
    // 叉积计算距离
    return std::abs((dy * px - dx * py + lx2 * ly1 - ly2 * lx1)) / length;
}

std::pair<double, double> MathUtils::projectPointToLine(
    double px, double py,
    double lx1, double ly1, double lx2, double ly2) {
    
    double dx = lx2 - lx1;
    double dy = ly2 - ly1;
    double length_sq = dx * dx + dy * dy;
    
    if (length_sq < 1e-10) {
        return {lx1, ly1};
    }
    
    double t = ((px - lx1) * dx + (py - ly1) * dy) / length_sq;
    return {lx1 + t * dx, ly1 + t * dy};
}
```

#### 曲率计算

```cpp
double MathUtils::curvatureFromThreePoints(
    double x1, double y1,
    double x2, double y2,
    double x3, double y3) {
    
    // 使用向量叉积计算有向面积
    double area = 0.5 * ((x2 - x1) * (y3 - y1) - (x3 - x1) * (y2 - y1));
    
    // 计算三边长
    double a = distance(x1, y1, x2, y2);
    double b = distance(x2, y2, x3, y3);
    double c = distance(x3, y3, x1, y1);
    
    double denominator = a * b * c;
    
    if (denominator < 1e-10) {
        return 0.0;  // 三点共线或重合
    }
    
    // 曲率 = 4 * 有向面积 / (a * b * c)
    return 4.0 * area / denominator;
}

Optional<std::pair<double, double>> MathUtils::circleCenter(
    double x1, double y1,
    double x2, double y2,
    double x3, double y3) {
    
    double d = 2.0 * (x1 * (y2 - y3) + x2 * (y3 - y1) + x3 * (y1 - y2));
    
    if (std::abs(d) < 1e-10) {
        return nullopt;  // 三点共线
    }
    
    double ux = ((x1*x1 + y1*y1) * (y2 - y3) + 
                 (x2*x2 + y2*y2) * (y3 - y1) +
                 (x3*x3 + y3*y3) * (y1 - y2)) / d;
    
    double uy = ((x1*x1 + y1*y1) * (x3 - x2) + 
                 (x2*x2 + y2*y2) * (x1 - x3) +
                 (x3*x3 + y3*y3) * (x2 - x1)) / d;
    
    return Optional<std::pair<double, double>>({ux, uy});
}
```

## 9.4 Optional 工具类

### 9.4.1 设计目的

由于项目使用 C++11 标准（`std::optional` 是 C++17 才有的），需要自定义 `Optional<T>` 类。

### 9.4.2 实现

```cpp
template<typename T>
class Optional {
public:
    Optional() : has_value_(false) {}
    
    Optional(const T& value) : has_value_(true), value_(value) {}
    
    Optional(T&& value) : has_value_(true), value_(std::move(value)) {}

    bool has_value() const { return has_value_; }
    
    explicit operator bool() const { return has_value_; }

    T& value() {
        if (!has_value_) {
            throw std::runtime_error("Optional has no value");
        }
        return value_;
    }
    
    const T& value() const {
        if (!has_value_) {
            throw std::runtime_error("Optional has no value");
        }
        return value_;
    }

    T value_or(const T& default_value) const {
        return has_value_ ? value_ : default_value;
    }

    T& operator*() { return value(); }
    const T& operator*() const { return value(); }
    
    T* operator->() { return &value(); }
    const T* operator->() const { return &value(); }

    void reset() { has_value_ = false; }

private:
    bool has_value_;
    T value_;
};

// 全局辅助变量
struct NulloptType {};
constexpr NulloptType nullopt{};

template<typename T>
Optional<T> make_optional(const T& value) {
    return Optional<T>(value);
}
```
