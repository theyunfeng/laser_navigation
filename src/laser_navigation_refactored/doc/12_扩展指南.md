# 12. 扩展指南

本章介绍如何扩展导航模块，包括添加新的轨迹类型、控制器和底盘类型。

## 12.1 添加新的轨迹类型

### 12.1.1 步骤概览

1. 在 `PathSegmentType` 枚举中添加新类型
2. 创建继承自 `PathSegment` 的新类
3. 在 `PathSegmentFactory` 中注册
4. 在 `JsonParser` 中添加解析支持
5. 在 `NavigationExecutor` 中添加跟踪逻辑

```cpp
/**
 * @brief Catmull-Rom 样条曲线段
 * 
 * 通过多个控制点生成平滑曲线，曲线会通过所有控制点。
 */
class CatmullRomSegment : public PathSegment {
public:
    /**
     * @brief 构造函数
     * @param control_points 控制点序列（至少4个）
     * @param constraints 运动约束
     * @param tension 张力参数（0~1，默认0.5）
     */
    CatmullRomSegment(const std::vector<Pose2D>& control_points,
                       const MotionConstraints& constraints,
                       double tension = 0.5);

    ~CatmullRomSegment() override = default;

    // PathSegment 接口实现
    PathSegmentType getType() const override { 
        return PathSegmentType::kCatmullRomSpline; 
    }
    
    Pose2D getStartPose() const override;
    Pose2D getEndPose() const override;
    double getStartHeading() const override;
    double getEndHeading() const override;
    double getTotalLength() const override { return total_length_; }
    MotionConstraints getConstraints() const override { return constraints_; }

    Eigen::Vector2d getPointAtParameter(double t) const override;
    double getHeadingAtParameter(double t) const override;
    double getCurvatureAtParameter(double t) const override;

    double getRemainingDistance(const Pose2D& current_pose) const override;
    double getLateralDeviation(const Pose2D& current_pose) const override;
    bool isReached(const Pose2D& current_pose, double threshold) const override;

    // 弧长参数化
    Eigen::Vector2d getPointAtArcLength(double s) const;
    double arcLengthToParameter(double s) const;

private:
    // Catmull-Rom 基函数
    Eigen::Vector2d evaluateSpline(double t, int segment_index) const;
    Eigen::Vector2d evaluateDerivative(double t, int segment_index) const;
    Eigen::Vector2d evaluateSecondDerivative(double t, int segment_index) const;
    
    void computeArcLengthTable();
    int findSegmentIndex(double global_t) const;

    std::vector<Pose2D> control_points_;
    MotionConstraints constraints_;
    double tension_;
    double total_length_;
    
    // 弧长查找表
    std::vector<double> arc_length_table_;
    static constexpr int kTableResolution = 100;
};

} // namespace laser_navigation_refactored
```

#### Step 3: 实现路径段

```cpp
// src/path/catmull_rom_segment.cpp

#include "laser_navigation_refactored/path/catmull_rom_segment.hpp"
#include "laser_navigation_refactored/core/math_utils.hpp"

namespace laser_navigation_refactored {

CatmullRomSegment::CatmullRomSegment(
    const std::vector<Pose2D>& control_points,
    const MotionConstraints& constraints,
    double tension)
    : control_points_(control_points)
    , constraints_(constraints)
    , tension_(tension) {
    
    if (control_points_.size() < 4) {
        throw std::invalid_argument(
            "Catmull-Rom spline requires at least 4 control points");
    }
    
    computeArcLengthTable();
}

Eigen::Vector2d CatmullRomSegment::evaluateSpline(
    double t, int segment_index) const {
    
    // Catmull-Rom 样条公式
    // P(t) = 0.5 * [(2*P1) + (-P0 + P2)*t + 
    //        (2*P0 - 5*P1 + 4*P2 - P3)*t^2 +
    //        (-P0 + 3*P1 - 3*P2 + P3)*t^3]
    
    const auto& p0 = control_points_[segment_index];
    const auto& p1 = control_points_[segment_index + 1];
    const auto& p2 = control_points_[segment_index + 2];
    const auto& p3 = control_points_[segment_index + 3];
    
    double t2 = t * t;
    double t3 = t2 * t;
    
    double x = 0.5 * ((2 * p1.x) +
                      (-p0.x + p2.x) * t +
                      (2*p0.x - 5*p1.x + 4*p2.x - p3.x) * t2 +
                      (-p0.x + 3*p1.x - 3*p2.x + p3.x) * t3);
    
    double y = 0.5 * ((2 * p1.y) +
                      (-p0.y + p2.y) * t +
                      (2*p0.y - 5*p1.y + 4*p2.y - p3.y) * t2 +
                      (-p0.y + 3*p1.y - 3*p2.y + p3.y) * t3);
    
    return Eigen::Vector2d(x, y);
}

void CatmullRomSegment::computeArcLengthTable() {
    arc_length_table_.clear();
    arc_length_table_.push_back(0.0);
    
    int num_segments = control_points_.size() - 3;
    double total_samples = kTableResolution * num_segments;
    
    Eigen::Vector2d prev_point = evaluateSpline(0, 0);
    double accumulated_length = 0.0;
    
    for (int i = 1; i <= total_samples; ++i) {
        double global_t = static_cast<double>(i) / total_samples;
        int seg_idx = findSegmentIndex(global_t);
        double local_t = (global_t * num_segments) - seg_idx;
        
        Eigen::Vector2d current_point = evaluateSpline(local_t, seg_idx);
        accumulated_length += (current_point - prev_point).norm();
        arc_length_table_.push_back(accumulated_length);
        prev_point = current_point;
    }
    
    total_length_ = accumulated_length;
}

// ... 其他方法实现

} // namespace laser_navigation_refactored
```

## 12.2 添加新的控制器

### 12.2.1 控制器接口

```cpp
// controller/controller_interface.hpp

namespace laser_navigation_refactored {

/**
 * @brief 控制器基类接口
 */
class ControllerInterface {
public:
    virtual ~ControllerInterface() = default;

    /// @brief 计算控制输出
    /// @param current_pose 当前位姿
    /// @param reference 参考（目标点、参考速度等）
    /// @param is_forward 是否前进
    /// @return 控制速度
    virtual Velocity compute(const Pose2D& current_pose,
                              const Pose2D& reference,
                              bool is_forward) = 0;

    /// @brief 带里程计反馈的控制
    virtual Velocity computeWithFeedback(
        const Pose2D& current_pose,
        const Pose2D& reference,
        bool is_forward,
        const Optional<OdometryFeedback>& odom) {
        // 默认忽略反馈
        return compute(current_pose, reference, is_forward);
    }

    /// @brief 设置底盘类型
    virtual void setChassisType(ChassisType type) { chassis_type_ = type; }

    /// @brief 重置控制器状态
    virtual void reset() {}

protected:
    ChassisType chassis_type_{ChassisType::kDifferential};
};

} // namespace laser_navigation_refactored
```

### 12.2.2 示例：添加 PID 控制器

```cpp
// controller/pid_controller.hpp

#pragma once

#include "laser_navigation_refactored/controller/controller_interface.hpp"

namespace laser_navigation_refactored {

struct PIDParams {
    double kp_distance{1.0};
    double ki_distance{0.0};
    double kd_distance{0.1};
    double kp_angle{2.0};
    double ki_angle{0.0};
    double kd_angle{0.1};
    double max_v{0.5};
    double max_w{0.5};
    double dt{0.04};
};

class PIDController : public ControllerInterface {
public:
    PIDController() = default;
    explicit PIDController(const PIDParams& params);

    void setParams(const PIDParams& params) { params_ = params; }

    Velocity compute(const Pose2D& current_pose,
                      const Pose2D& reference,
                      bool is_forward) override;

    void reset() override;

private:
    PIDParams params_;
    double prev_distance_error_{0.0};
    double prev_angle_error_{0.0};
    double integral_distance_{0.0};
    double integral_angle_{0.0};
};

// 实现
Velocity PIDController::compute(
    const Pose2D& current_pose,
    const Pose2D& reference,
    bool is_forward) {
    
    // 距离误差
    double dx = reference.x - current_pose.x;
    double dy = reference.y - current_pose.y;
    double distance_error = std::sqrt(dx * dx + dy * dy);

    // 角度误差
    double target_angle = std::atan2(dy, dx);
    if (!is_forward) {
        target_angle = math::MathUtils::normalizeAngle(target_angle + M_PI);
    }
    double angle_error = math::MathUtils::angleDiff(
        target_angle, current_pose.yaw);

    // PID 计算 - 线速度
    double d_distance = (distance_error - prev_distance_error_) / params_.dt;
    integral_distance_ += distance_error * params_.dt;
    double v = params_.kp_distance * distance_error +
               params_.ki_distance * integral_distance_ +
               params_.kd_distance * d_distance;

    // PID 计算 - 角速度
    double d_angle = (angle_error - prev_angle_error_) / params_.dt;
    integral_angle_ += angle_error * params_.dt;
    double w = params_.kp_angle * angle_error +
               params_.ki_angle * integral_angle_ +
               params_.kd_angle * d_angle;

    // 更新历史
    prev_distance_error_ = distance_error;
    prev_angle_error_ = angle_error;

    // 限幅
    v = std::clamp(v, -params_.max_v, params_.max_v);
    w = std::clamp(w, -params_.max_w, params_.max_w);

    // 倒车处理
    if (!is_forward) {
        v = -std::abs(v);
    }

    return Velocity(v, 0.0, w);
}

} // namespace laser_navigation_refactored
```

### 12.3.1 步骤

4. 更新 JSON 解析

### 12.3.2 示例：添加麦克纳姆轮底盘

```cpp
// types.hpp
enum class ChassisType {
    kDifferential,
    kSwerve4WIS4WID,
    kMecanum  // 新增
};

// json_parser.hpp
ChassisType JsonParser::parseChassisType(const nlohmann::json& json) {
    if (!json.contains("chassis_type")) {
        return ChassisType::kDifferential;
    }
    
    std::string type_str = json["chassis_type"].get<std::string>();
    
    if (type_str == "Swerve4WIS4WID" || type_str == "4WIS4WID") {
        return ChassisType::kSwerve4WIS4WID;
    }
    if (type_str == "Mecanum") {
        return ChassisType::kMecanum;
    }
    
    return ChassisType::kDifferential;
}

// navigation_executor.cpp
Velocity NavigationExecutor::adjustVelocityForChassis(
    const Velocity& velocity) const {
    
    Velocity adjusted = velocity;
    
    switch (chassis_type_) {
        case ChassisType::kDifferential:
            // 差速底盘不支持横向速度
            adjusted.linear_y = 0.0;
            break;
            
        case ChassisType::kSwerve4WIS4WID:
        case ChassisType::kMecanum:
            // 全向底盘支持横向速度
            break;
    }
    
    return adjusted;
}
```

### 12.5.2 示例：梯形速度规划器

```cpp
// planner/trapezoidal_planner.hpp

class TrapezoidalPlanner : public VelocityPlannerInterface {
public:
    void initialize(double total_distance,
                    double initial_velocity,
                    const MotionConstraints& constraints) override {
        total_distance_ = total_distance;
        current_velocity_ = initial_velocity;
        max_velocity_ = constraints.max_velocity;
        acceleration_ = constraints.max_acceleration;
        deceleration_ = constraints.max_deceleration;
        
        // 计算各阶段距离
        computePhases();
    }

    double computeVelocity(double remaining_distance, double dt) override {
        double traveled = total_distance_ - remaining_distance;
        
        if (traveled < accel_distance_) {
            // 加速阶段
            current_velocity_ = std::min(
                current_velocity_ + acceleration_ * dt,
                max_velocity_);
        } else if (remaining_distance < decel_distance_) {
            // 减速阶段
            double v_target = std::sqrt(2 * deceleration_ * remaining_distance);
            current_velocity_ = std::max(
                current_velocity_ - deceleration_ * dt,
                v_target);
        }
        // 匀速阶段保持 current_velocity_
        
        return current_velocity_;
    }

private:
    void computePhases() {
        // 计算达到最大速度所需距离
        accel_distance_ = (max_velocity_ * max_velocity_) / (2 * acceleration_);
        
        // 计算从最大速度减到零所需距离
        decel_distance_ = (max_velocity_ * max_velocity_) / (2 * deceleration_);
        
        // 如果距离不够，调整最大速度
        if (accel_distance_ + decel_distance_ > total_distance_) {
            max_velocity_ = std::sqrt(
                2 * total_distance_ * acceleration_ * deceleration_ /
                (acceleration_ + deceleration_));
            accel_distance_ = (max_velocity_ * max_velocity_) / (2 * acceleration_);
            decel_distance_ = (max_velocity_ * max_velocity_) / (2 * deceleration_);
        }
    }

    double total_distance_;
    double current_velocity_;
    double max_velocity_;
    double acceleration_;
    double deceleration_;
    double accel_distance_;
    double decel_distance_;
};
```

## 12.8 最佳实践

1. **接口优先**: 先定义清晰的接口，再实现具体功能
2. **单一职责**: 每个类只负责一个功能
3. **可测试性**: 设计时考虑单元测试
4. **文档**: 为新增的公共接口添加 Doxygen 注释
5. **兼容性**: 保持 C++11 兼容，不使用 C++14/17 特性
