# 11. 使用示例

本章提供完整的代码示例，帮助快速集成导航模块。

## 11.1 ROS 2 节点示例

### 11.1.1 完整节点代码

```cpp
// navigation_example_node.cpp

#include <rclcpp/rclcpp.hpp>
#include <geometry_msgs/msg/twist.hpp>
#include <geometry_msgs/msg/pose_stamped.hpp>
#include <nav_msgs/msg/odometry.hpp>
#include <std_msgs/msg/string.hpp>
#include <std_msgs/msg/bool.hpp>

#include "laser_navigation_refactored/navigation_executor.hpp"
#include "laser_navigation_refactored/utils/json_parser.hpp"

using namespace laser_navigation_refactored;

class NavigationExampleNode : public rclcpp::Node {
public:
    NavigationExampleNode() : Node("navigation_example") {
        // 参数声明
        this->declare_parameter("control_period", 0.04);
        this->declare_parameter("model_config_path", "model.json");
        this->declare_parameter("lqr_q_x", 10.0);
        this->declare_parameter("lqr_q_y", 10.0);
        this->declare_parameter("lqr_q_theta", 5.0);
        this->declare_parameter("lqr_r_v", 1.0);
        this->declare_parameter("lqr_r_w", 1.0);

        // 获取参数
        control_period_ = this->get_parameter("control_period").as_double();
        
        // 从 model.json 读取底盘类型
        std::string model_path = this->get_parameter("model_config_path").as_string();
        loadChassisType(model_path);

        // 初始化 LQR 参数
        initLQRParams();

        // 创建发布者
        cmd_vel_pub_ = this->create_publisher<geometry_msgs::msg::Twist>(
            "cmd_vel", 10);
        status_pub_ = this->create_publisher<std_msgs::msg::String>(
            "navigation_status", 10);

        // 创建订阅者
        odom_sub_ = this->create_subscription<nav_msgs::msg::Odometry>(
            "odom", 10,
            std::bind(&NavigationExampleNode::odomCallback, this,
                      std::placeholders::_1));

        task_sub_ = this->create_subscription<std_msgs::msg::String>(
            "navigation_task", 10,
            std::bind(&NavigationExampleNode::taskCallback, this,
                      std::placeholders::_1));

        obstacle_sub_ = this->create_subscription<std_msgs::msg::Bool>(
            "obstacle_detected", 10,
            std::bind(&NavigationExampleNode::obstacleCallback, this,
                      std::placeholders::_1));

        cancel_sub_ = this->create_subscription<std_msgs::msg::Bool>(
            "cancel_navigation", 10,
            std::bind(&NavigationExampleNode::cancelCallback, this,
                      std::placeholders::_1));

        // 创建控制定时器
        control_timer_ = this->create_wall_timer(
            std::chrono::milliseconds(static_cast<int>(control_period_ * 1000)),
            std::bind(&NavigationExampleNode::controlLoop, this));

        // 设置日志回调
        executor_.setLogCallback([this](const std::string& msg) {
            RCLCPP_INFO(this->get_logger(), "%s", msg.c_str());
        });

        RCLCPP_INFO(this->get_logger(), "Navigation node initialized");
    }

private:
    void initLQRParams() {
        lqr_params_.dt = control_period_;
        lqr_params_.Q_x = this->get_parameter("lqr_q_x").as_double();
        lqr_params_.Q_y = this->get_parameter("lqr_q_y").as_double();
        lqr_params_.Q_theta = this->get_parameter("lqr_q_theta").as_double();
        lqr_params_.R_v = this->get_parameter("lqr_r_v").as_double();
        lqr_params_.R_w = this->get_parameter("lqr_r_w").as_double();
        lqr_params_.max_v = 1.0;
        lqr_params_.max_w = 1.5;

        bezier_lqr_params_.dt = control_period_;
        bezier_lqr_params_.Q_x = 15.0;
        bezier_lqr_params_.Q_y = 15.0;
        bezier_lqr_params_.Q_theta = 8.0;
        bezier_lqr_params_.R_v = 1.0;
        bezier_lqr_params_.R_w = 1.0;
    }

    void loadChassisType(const std::string& model_path) {
        // 从 model.json 读取 agvType
        JsonParser parser;
        chassis_type_ = parser.parseModelConfig(model_path);
        
        RCLCPP_INFO(this->get_logger(), "Chassis type: %s",
            chassis_type_ == ChassisType::kSwerve4WIS4WID ? 
            "4WIS4WID" : "DiffDrive");
    }

    void odomCallback(const nav_msgs::msg::Odometry::SharedPtr msg) {
        // 提取位姿
        current_pose_.x = msg->pose.pose.position.x;
        current_pose_.y = msg->pose.pose.position.y;

        // 四元数转欧拉角
        double siny_cosp = 2.0 * (msg->pose.pose.orientation.w *
                                   msg->pose.pose.orientation.z);
        double cosy_cosp = 1.0 - 2.0 * (msg->pose.pose.orientation.z *
                                         msg->pose.pose.orientation.z);
        current_pose_.yaw = std::atan2(siny_cosp, cosy_cosp);

        // 提取速度
        Velocity odom_vel;
        odom_vel.linear_x = msg->twist.twist.linear.x;
        odom_vel.linear_y = msg->twist.twist.linear.y;
        odom_vel.angular_z = msg->twist.twist.angular.z;

        // 更新里程计反馈
        double timestamp = this->now().seconds();
        executor_.updateOdometryFeedback(current_pose_, odom_vel, timestamp);

        pose_received_ = true;
    }

    void taskCallback(const std_msgs::msg::String::SharedPtr msg) {
        RCLCPP_INFO(this->get_logger(), "Received navigation task");

        // 解析 JSON
        JsonParser parser;
        auto config = parser.parseNavigationConfig(msg->data);

        if (!config.has_value()) {
            RCLCPP_ERROR(this->get_logger(), "Failed to parse task: %s",
                         parser.getLastError().c_str());
            return;
        }

        // 设置底盘类型（从 model.json 读取的）
        config.value().chassis_type = chassis_type_;

        if (!config.has_value()) {
            RCLCPP_ERROR(this->get_logger(), "Failed to parse task: %s",
                         parser.getLastError().c_str());
            return;
        }

        // 初始化执行器
        int ret = executor_.initialize(config.value(), lqr_params_,
                                        bezier_lqr_params_);
        if (ret != 0) {
            RCLCPP_ERROR(this->get_logger(), "Failed to initialize executor");
            return;
        }

        is_navigating_ = true;
        RCLCPP_INFO(this->get_logger(), "Navigation started with %zu waypoints",
                    config.value().waypoints.size());
    }

    void obstacleCallback(const std_msgs::msg::Bool::SharedPtr msg) {
        obstacle_detected_ = msg->data;
    }

    void cancelCallback(const std_msgs::msg::Bool::SharedPtr msg) {
        if (msg->data) {
            cancel_requested_ = true;
            RCLCPP_WARN(this->get_logger(), "Navigation cancel requested");
        }
    }

    void controlLoop() {
        if (!is_navigating_ || !pose_received_) {
            return;
        }

        // 执行导航
        double current_time = this->now().seconds();
        NavigationOutput output = executor_.executeWithOdometry(
            current_pose_, obstacle_detected_, 0.5,
            cancel_requested_, current_time);

        // 发布速度命令
        geometry_msgs::msg::Twist cmd_vel;
        cmd_vel.linear.x = output.velocity.linear_x;
        cmd_vel.linear.y = output.velocity.linear_y;
        cmd_vel.angular.z = output.velocity.angular_z;
        cmd_vel_pub_->publish(cmd_vel);

        // 发布状态
        publishStatus(output);

        // 检查是否完成
        if (output.status == NavigationStatus::kGoalReached ||
            output.status == NavigationStatus::kCancelled ||
            output.status == NavigationStatus::kError) {

            is_navigating_ = false;
            cancel_requested_ = false;

            // 停车
            geometry_msgs::msg::Twist stop_cmd;
            cmd_vel_pub_->publish(stop_cmd);

            RCLCPP_INFO(this->get_logger(), "Navigation ended: %s",
                        statusToString(output.status).c_str());
        }
    }

    void publishStatus(const NavigationOutput& output) {
        std_msgs::msg::String status_msg;
        status_msg.data = statusToString(output.status) +
            " | Segment: " + std::to_string(output.current_segment) +
            " | Remaining: " + std::to_string(output.remaining_distance) + "m";
        status_pub_->publish(status_msg);
    }

    std::string statusToString(NavigationStatus status) {
        switch (status) {
            case NavigationStatus::kIdle: return "IDLE";
            case NavigationStatus::kRunning: return "RUNNING";
            case NavigationStatus::kReversing: return "REVERSING";
            case NavigationStatus::kRotating: return "ROTATING";
            case NavigationStatus::kSegmentReached: return "SEGMENT_REACHED";
            case NavigationStatus::kGoalReached: return "GOAL_REACHED";
            case NavigationStatus::kObstaclePaused: return "OBSTACLE_PAUSED";
            case NavigationStatus::kCancelled: return "CANCELLED";
            case NavigationStatus::kError: return "ERROR";
            default: return "UNKNOWN";
        }
    }

    // 成员变量
    NavigationExecutor executor_;
    LQRParams lqr_params_;
    BezierLQRParams bezier_lqr_params_;
    ChassisType chassis_type_{ChassisType::kDifferential};  // 从 model.json 读取

    Pose2D current_pose_;
    bool pose_received_{false};
    bool is_navigating_{false};
    bool obstacle_detected_{false};
    bool cancel_requested_{false};
    double control_period_;

    // ROS 接口
    rclcpp::Publisher<geometry_msgs::msg::Twist>::SharedPtr cmd_vel_pub_;
    rclcpp::Publisher<std_msgs::msg::String>::SharedPtr status_pub_;
    rclcpp::Subscription<nav_msgs::msg::Odometry>::SharedPtr odom_sub_;
    rclcpp::Subscription<std_msgs::msg::String>::SharedPtr task_sub_;
    rclcpp::Subscription<std_msgs::msg::Bool>::SharedPtr obstacle_sub_;
    rclcpp::Subscription<std_msgs::msg::Bool>::SharedPtr cancel_sub_;
    rclcpp::TimerBase::SharedPtr control_timer_;
};

int main(int argc, char** argv) {
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<NavigationExampleNode>());
    rclcpp::shutdown();
    return 0;
}
```

## 11.2 非 ROS 使用示例

### 11.2.1 基本使用

```cpp
#include "laser_navigation_refactored/navigation_executor.hpp"
#include "laser_navigation_refactored/utils/json_parser.hpp"

int main() {
    using namespace laser_navigation_refactored;

    // 创建执行器
    NavigationExecutor executor;

    // 配置 LQR 参数
    LQRParams lqr_params;
    lqr_params.dt = 0.04;
    lqr_params.Q_x = 10.0;
    lqr_params.Q_y = 10.0;
    lqr_params.Q_theta = 5.0;
    lqr_params.R_v = 1.0;
    lqr_params.R_w = 1.0;
    lqr_params.max_v = 0.5;
    lqr_params.max_w = 0.5;

    BezierLQRParams bezier_params;
    bezier_params.dt = 0.04;
    bezier_params.Q_x = 15.0;
    bezier_params.Q_y = 15.0;
    bezier_params.Q_theta = 8.0;
    bezier_params.R_v = 1.0;
    bezier_params.R_w = 1.0;

    // 方式一：手动构建配置
    NavigationConfig config;
    config.chassis_type = ChassisType::kDifferential;
    config.adjust_start_angle = true;
    config.adjust_end_angle = true;

    config.waypoints = {
        {0.0, 0.0, 0.0},
        {2.0, 0.0, 0.0},
        {2.0, 2.0, 1.57}
    };

    config.segment_types = {
        PathSegmentType::kStraight,
        PathSegmentType::kCubicBezier
    };

    config.constraints = {
        {0.5, 0.3, 0.3, 0.5, 0.3, true, 0.05, 0.05},
        {0.3, 0.3, 0.3, 0.5, 0.3, true, 0.05, 0.05}
    };

    config.control_points = {
        {},  // 直线段无控制点
        {{2.5, 0.0, 0.0}, {2.0, 1.5, 0.0}}  // 贝塞尔控制点
    };

    // 初始化
    int ret = executor.initialize(config, lqr_params, bezier_params);
    if (ret != 0) {
        std::cerr << "Initialization failed" << std::endl;
        return -1;
    }

    // 控制循环
    Pose2D current_pose = {0.0, 0.0, 0.0};
    while (true) {
        // 从传感器获取当前位姿
        // current_pose = getSensorPose();

        // 执行导航
        NavigationOutput output = executor.execute(
            current_pose,
            false,  // 无障碍物
            0.5,    // 减速度
            false   // 不取消
        );

        // 发送速度命令
        std::cout << "Velocity: v=" << output.velocity.linear_x
                  << ", w=" << output.velocity.angular_z << std::endl;

        // 检查是否完成
        if (output.status == NavigationStatus::kGoalReached) {
            std::cout << "Goal reached!" << std::endl;
            break;
        }

        if (output.status == NavigationStatus::kError) {
            std::cerr << "Navigation error!" << std::endl;
            break;
        }

        // 模拟控制周期
        std::this_thread::sleep_for(std::chrono::milliseconds(40));
    }

    return 0;
}
```

### 11.3.3 障碍物处理

```cpp
bool obstacle_detected = false;

void obstacleDetectedCallback(bool detected) {
    obstacle_detected = detected;
}

void controlLoop() {
    NavigationOutput output = executor.execute(
        current_pose,
        obstacle_detected,  // 障碍物标志
        0.5,                // 减速度
        false
    );

    switch (output.status) {
        case NavigationStatus::kObstaclePaused:
            // 机器人已停止，等待障碍物消失
            std::cout << "Paused for obstacle" << std::endl;
            break;

        case NavigationStatus::kRunning:
            // 正常运行
            if (output.recovery_strategy != RecoveryStrategy::kNone) {
                // 刚从停止状态恢复
                switch (output.recovery_strategy) {
                    case RecoveryStrategy::kContinue:
                        std::cout << "Continuing navigation" << std::endl;
                        break;
                    case RecoveryStrategy::kReplan:
                        std::cout << "Replanning from current position" << std::endl;
                        break;
                    case RecoveryStrategy::kAbort:
                        std::cout << "Aborting due to large deviation" << std::endl;
                        break;
                }
            }
            break;
    }
}
```

## 11.4 命令行发送任务

使用 ROS 2 CLI 发送导航任务（底盘类型已在节点启动时从 model.json 读取）：

```bash
# 简单直线任务（不需要指定 chassis_type）
ros2 topic pub --once /navigation_task std_msgs/msg/String "data: '{
    \"waypoints\": [
        {\"x\": 0.0, \"y\": 0.0, \"yaw\": 0.0, \"type\": \"Straight\"},
        {\"x\": 2.0, \"y\": 0.0, \"yaw\": 0.0}
    ]
}'"

# 取消导航
ros2 topic pub --once /cancel_navigation std_msgs/msg/Bool "data: true"

# 模拟障碍物
ros2 topic pub /obstacle_detected std_msgs/msg/Bool "data: true"
```
